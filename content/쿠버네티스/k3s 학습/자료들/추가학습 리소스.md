#!/bin/bash

# Scale Out 테스트를 위한 부하 생성 스크립트

MASTER_IP=${1:-"localhost"}
GATEWAY_PORT=${2:-"30080"}
DURATION=${3:-60}  # 초
CONCURRENT=${4:-10}  # 동시 연결 수

echo "=== Load Test Configuration ==="
echo "Target: http://${MASTER_IP}:${GATEWAY_PORT}"
echo "Duration: ${DURATION} seconds"
echo "Concurrent connections: ${CONCURRENT}"
echo "================================"
echo

# HPA 초기 상태 확인
echo "Initial HPA status:"
kubectl get hpa
echo

# 부하 생성 함수
generate_load() {
    local endpoint=$1
    local count=0
    local start_time=$(date +%s)
    
    while [ $(($(date +%s) - start_time)) -lt $DURATION ]; do
        response=$(curl -s -w "\n%{http_code}" http://${MASTER_IP}:${GATEWAY_PORT}${endpoint})
        http_code=$(echo "$response" | tail -n1)
        
        if [ "$http_code" = "200" ]; then
            ((count++))
        fi
        
        # 간단한 프로그레스 표시
        if [ $((count % 100)) -eq 0 ]; then
            echo "[$(date +%T)] Endpoint: ${endpoint}, Requests sent: ${count}"
        fi
    done
    
    echo "Total successful requests to ${endpoint}: ${count}"
}

# 백그라운드로 여러 엔드포인트에 동시 부하 생성
echo "Starting load test..."
for i in $(seq 1 $CONCURRENT); do
    if [ $((i % 3)) -eq 0 ]; then
        generate_load "/users" &
    elif [ $((i % 3)) -eq 1 ]; then
        generate_load "/products" &
    else
        generate_load "/" &
    fi
done

# HPA 상태 모니터링 (별도 백그라운드)
(
    while true; do
        clear
        echo "=== Real-time Monitoring ==="
        echo "Time: $(date +%T)"
        echo
        echo "--- HPA Status ---"
        kubectl get hpa
        echo
        echo "--- Pod Count ---"
        kubectl get pods -l app=gateway -o wide | head -n 10
        echo
        echo "--- CPU/Memory Usage ---"
        kubectl top pods -l app=gateway 2>/dev/null || echo "Metrics not available yet"
        sleep 5
    done
) &
MONITOR_PID=$!

# 모든 부하 생성이 완료될 때까지 대기
wait

# 모니터링 프로세스 종료
kill $MONITOR_PID 2>/dev/null

echo
echo "=== Load Test Complete ==="
echo "Final HPA status:"
kubectl get hpa
echo
echo "Final Pod distribution:"
kubectl get pods -l app=gateway -o wide
echo
echo "Waiting for scale down... (check with: watch kubectl get hpa)"
