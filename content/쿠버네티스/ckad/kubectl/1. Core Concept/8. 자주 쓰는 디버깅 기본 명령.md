
```bash
# 상태
k get pod                          # 목록과 상태
k describe pod my-pod              # 상세 + 이벤트

# 로그
k logs my-pod
k logs my-pod --previous

# 안에서 명령
k exec my-pod -- ls /app
k exec -it my-pod -- sh

# 임시 디버깅 Pod
k run tmp --image=busybox --rm -it --restart=Never -- sh
k run curl --image=curlimages/curl --rm -it --restart=Never -- sh

# 이벤트
k get events --sort-by=.metadata.creationTimestamp
```
